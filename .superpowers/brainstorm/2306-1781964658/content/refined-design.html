板块 1：可动画属性 —— 强类型系统

// 每种属性有明确的类型，编译器强制检查

// ── 数值 (float) ──
opacity: 0→1           // 0.0 ~ 1.0
scale / scaleX / scaleY / scaleZ: 1→1.5

// ── 长度 (length) ──
x / y / translateZ: 0→100px | 0%→50%
width / height: 100px→200px
borderRadius: 0→20px
margin / padding: 0→20px

// ── 角度 (angle) ──
rotate / rotateX / rotateY / rotateZ: 0deg→360deg
skewX / skewY: 0deg→10deg

// ── 颜色 (color) ──
color / backgroundColor / borderColor: #ff0000→#00ff00
fill / stroke: #000→#fff

// ── SVG 专用 ──
strokeWidth: 1→5
strokeDashoffset: 100%→0
r: 10→20
d: path("M0,0 L10,10")→path("M0,0 L20,20")

// ❌ display: none→block      → 编译错误：display 不可插值
// ❌ opacity: 0→"hello"       → 编译错误：类型不匹配
// ❌ width: auto→300px        → 编译错误：auto 不可插值，请用具体值
// ⚠️  boxShadow: ...→...     → 警告：Flutter 暂不支持，Web 可用

板块 2：Stagger —— 两种模式都支持

// 模式 A：间隔模式（默认）
// each: 60ms → 每个 item 比上一个延迟 60ms
stagger(motion, each: 60ms)
// 总时长 = (N-1) * 60ms + motion.duration

// 模式 B：分布模式
// amount: 600ms → 所有 item 在 600ms 内均匀分布
stagger(motion, amount: 600ms)
// 每个间隔 = 600ms / (N-1)
// 总时长 = 600ms + motion.duration

// from 精确语义：
// start:   item[0] 先播 → item[N-1] 最后
// center:  item[N/2] 先播 → 向两边扩散
// end:     item[N-1] 先播 → item[0] 最后
// edges:   item[0] 和 item[N-1] 同时先播 → 向中间收缩

stagger(motion, each: 60ms, from: center)
stagger(motion, amount: 600ms, from: edges)

板块 3：时间线语法 —— GSAP 参考版

// GSAP 的 position 参数理念：
//   ">"  = 上一个结束后
//   "<"  = 和上一个同时开始
//   "+=N" = 上一个结束后 + N
//   "-=N" = 上一个结束前 N（重叠）
//   绝对时间 = 从时间线起点算

// OmniView 版本——用 start 参数表达同样的语义：
@motion entrance {
  // 默认 start: ">"（上一个结束后）
  fadeIn  from { opacity: 0 } over: 300ms

  // start: ">" 显式写
  slideUp from { y: 30 } over: 200ms start: ">"

  // start: "-=0.1s"（上一个结束前 0.1s 开始 = 重叠）
  scaleUp to { scale: 1.05 } over: 150ms start: "-=0.1s"

  // start: "<"（和上一个同时开始）
  glow to { opacity: 0.7 } over: 200ms start: "<"

  // start: "<+0.3s"（上一个开始后 0.3s）
  bounce to { scale: 1 } over: 200ms start: "<+0.3s"

  // start: 绝对时间
  spark to { opacity: 0 } over: 100ms start: 1.5s
}

// 等价于 GSAP：
// tl.to(el, { opacity: 1, duration: 0.3 })
//   .to(el, { y: 30, duration: 0.2 })
//   .to(el, { scale: 1.05, duration: 0.15 }, "-=0.1")
//   .to(el, { opacity: 0.7, duration: 0.2 }, "<")
//   .to(el, { scale: 1, duration: 0.2 }, "<+0.3")
//   .to(el, { opacity: 0, duration: 0.1 }, 1.5)

板块 4：@hover —— 两种反向模式

// 模式 A：smooth（默认）—— 从当前进度倒播
// 总在正在进行的动画的当前进度处反向
<button @hover={scale(1.05)}>
// 等价于 @hover={scale(1.05)} @hover-reverse={smooth}

// 模式 B：snap —— 立即跳到终点，然后播放离开动画
<button @hover={scaleUp} @hover-reverse={snap}>
// 等价于 hover 进入→scaleUp，hover 离开→瞬间归位

// 如果需要完全不一样的离开动画：
<button @hover={scaleUp} @unhover={scaleDown}>

板块 5：动画引用 API

// 通过触发词引用动画
<div ref:card @enter={fadeIn} @leave={fadeOut} @hover={glow}>

action demo {
  // 默认动画就是触发词
  card.enter.play()
  card.enter.pause()
  card.enter.resume()
  card.enter.reverse()
  card.enter.seek(0.5)        // 跳到 50% 进度
  card.enter.kill()            // 终止并清除

  // 等待动画完成
  await card.enter.complete    // Promise / Future

  // 只读状态
  card.enter.progress          // 0→1
  card.enter.playing           // bool
  card.enter.completed         // bool
}

// 如果一个触发词有多个动画（如 stagger），返回数组
card.enter[0].progress

板块 6：@driven-by —— 参考 GSAP progress()

// 等价于每帧调用 animation.progress(clamp(expr, 0, 1))
<div @animate={fadeIn} @driven-by={scrollProgress}>

// 行为：
//   expr ≤ 0 → progress = 0（停在起点）
//   0 < expr < 1 → progress = expr
//   expr ≥ 1 → progress = 1（停在终点）
//   expr 往回走 → progress 跟随往回（可倒放）

// 多个动画共享一个 driven-by：
<div
  @animate={fadeIn}
  @animate={slideUp}
  @driven-by={scrollProgress}>

// 动画不自主播放——完全由 driven-by 控制
// loop / direction 在 driven-by 模式下被忽略

板块 7：关键帧——GSAP 风格：每个 stop 可独立 ease

// 共享 over，每段可独立 ease
@motion bounce {
  0%   { y: 0 }     ease: easeOut
  40%  { y: -30px } ease: easeIn
  60%  { y: -15px } ease: easeOut
  100% { y: 0 }
  over: 600ms
}

// 如果只写一个 ease，所有段共享
@motion fadeSlide {
  0%   { opacity: 0; x: -20px }
  100% { opacity: 1; x: 0 }
  over: 300ms ease: easeOut        // 全局 ease
}

// from/to 是 0%/100% 的语法糖
@motion simple {
  from { opacity: 0 }        // 等价于 0% { opacity: 0 }
  to   { opacity: 1 }        // 等价于 100% { opacity: 1 }
  over: 300ms ease: easeOut
}
