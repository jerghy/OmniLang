统一冒号语法 & 完整设计总览

所有属性 key: value，数字带单位，和 CSS 风格一致

Motion 块完整语法

@motion cardEnter {
  // ── 帧 ──
  from { opacity: 0; y: 30px }
  to   { opacity: 1; y: 0 }

  // ── 时序 ──
  over: 400ms
  ease: easeOut
  delay: 0ms

  // ── 播放 ──
  loop: infinite       // number | infinite
  direction: alternate // normal | reverse | alternate | alternateReverse

  // ── 填充 ──
  fill: both           // none | forwards | backwards | both
}

@motion bounce {
  0%   { y: 0 }
  40%  { y: -30px }
  60%  { y: -15px }
  100% { y: 0 }
  over: 600ms
  ease: easeOut
}

组合 & 挂载

// ── 顺序 / 并行 / 重叠 ──
@motion entrance {
  from { opacity: 0; y: 30px } over: 400ms ease: easeOut
  then from { opacity: 0; x: -20px } over: 300ms
  then stagger(
    from { opacity: 0; y: 10px } over: 300ms ease: back(intensity: 1.7),
    each: 60ms,
    from: center
  ) at: -0.1s
}

// ── 元素挂载 ──
<div @enter={entrance} @leave={fadeOut}>
<button @hover={scale(1.05)} @press={scale(0.95)}>
<ul @enter={stagger(fadeIn, each: 60ms, from: center)}>

// ── 内联 ──
<div @enter={to { opacity: 1; y: 0 } over: 300ms ease: easeOut}>

// ── 信号绑定 ──
<div @bind-style={cardMotion} @bind-signal={isOpen}>

// ── 滚动 ──
<section @scroll={fadeIn} @scroll-trigger={top enters 80%}>

// ── 手势 ──
<Card @drag-horizontal={swipeCard} @drag-threshold={30%}>

// ── 多状态 ──
<Panel @states={{
  idle:    to { scale: 1;   opacity: 1 }   over: 150ms,
  hover:   to { scale: 1.05; opacity: 1 }   over: 150ms,
  pressed: to { scale: 0.95; opacity: 0.8 } over: 100ms
}}>

完整设计决策汇总

层 决策 语法
定义 命名块 @motion NAME { ... }
帧 from/to + 百分比混用，无包裹关键字 from { ... } to { ... } / 40% { ... }
方向 3 种原语 to {} / from {} / from {} to {}
属性 统一 key: value + 冒号 over: 300ms ease: easeOut
数值 显式单位 300ms 1.5s 20px 30%
缓动 预设 + 贝塞尔 + 物理弹簧 ease: spring(tension: 200, friction: 15)
组合 关键字链式 then / with / at: -0.1s
交错 函数修饰符 stagger(motion, each: 60ms, from: center)
挂载 表达式属性（和 @onClick 一致） @enter={expr} @leave={expr} @hover={expr}
触发词 6 类 enter / leave / hover / press / focus / drag / scroll / animate / bind / states
