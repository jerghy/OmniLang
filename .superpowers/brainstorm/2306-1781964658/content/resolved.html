已解决的问题

这些有明确答案，直接定下来

1. 3D 变换

// 新增属性（Web: CSS 3D transforms / Flutter: Transform widget 3D）
rotateX / rotateY / rotateZ
translateZ
perspective      // 透视距离
scaleZ

@motion flipCard {
  to { rotateY: 180deg } over: 600ms ease: easeInOut
  perspective: 1000px
}

2. SVG 动画属性

// SVG 独有属性（Web: CSS/WAAPI / Flutter: CustomPaint）
strokeDashoffset    // 线条绘制
fill / stroke       // 填充色 / 描边色
strokeWidth         // 描边宽度
r / cx / cy         // 圆
d                   // path 数据（形状变换）

@motion drawCheck {
  from { strokeDashoffset: 100%; opacity: 0 }
  to   { strokeDashoffset: 0;    opacity: 1 }
  over: 500ms ease: easeOut
}

3. timeScale —— 全局和单动画速度控制

// 全局（编译到 CSS 变量 / Flutter Theme）
--omni-motion-speed: 0.5        // CSS 变量，0=暂停 1=正常 2=双倍

// 单动画
@motion fastPulse {
  to { scale: 1.1 } over: 300ms
  timeScale: 2.0                // 2 倍速
}

4. will-change 自动管理

// 编译器行为，无需语法
// 当动画目标属性包含 transform / opacity 时：
//   → 动画开始前自动添加 will-change: transform, opacity
//   → 动画结束后 500ms 自动移除（防止内存泄漏）
// 可通过 @motion 关闭
@motion fadeIn {
  to { opacity: 1 } over: 300ms
  willChange: false           // 手动关闭自动 will-change
}

5. prefers-reduced-motion

// 默认遵守系统设置，可覆盖
@motion fadeIn {
  to { opacity: 1 } over: 300ms
  respectReducedMotion: true    // 默认 true。减少时 → over: 0ms，瞬间完成
}

// Web: matchMedia('(prefers-reduced-motion: reduce)')
// Flutter: MediaQuery.of(context).disableAnimations

6. @in-view 触发 —— 滚动到视野时播放

// 不同于 @scroll（scrub 模式），@in-view 是触发式
<div @in-view={fadeIn}>                          // 进入视野播放
<div @in-view={fadeIn} @in-view-once={true}>     // 只播一次
<div @in-view={fadeIn} @in-view-repeat={true}>   // 每次进入都播
<div @in-view={fadeIn} @in-view-threshold={0.5}>  // 50% 可见时触发

// Web: IntersectionObserver
// Flutter: VisibilityDetector 或 ScrollController

7. 动态值语义 —— 动画启动时快照

// 动画参数在 start 时刻快照，中途信号变化不更新
<div @enter={to { y: {targetY} } over: {speed}ms}>
// targetY=100, speed=300 → 动画在 t=0 确定
// 动画过程中 targetY 变 200 → 忽略

// 如果需要追踪动态值，用 @driven-by
<div @animate={slideX} @driven-by={scrollProgress}>

8. 相对值 += / -=

motion hoverScale {
  to { scale: +=0.05 } over: 150ms    // 当前值 + 0.05
  // 等价于 GSAP: gsap.to(el, { scale: "+=0.05" })
}

// 也支持 -= / *=
to { x: -=20px }
to { scale: *=1.1 }

9. 中断策略

@motion fadeIn {
  to { opacity: 1 } over: 300ms
  interrupt: stop        // stop: 停住当前值（默认）
                         // complete: 立刻跳到终点
                         // queue: 排队等上一个播完
                         // replace: 新动画直接替换旧动画
}

10. repeatRefresh —— 每次循环重新取 from 值

@motion sparkle {
  from { x: {randomX()}; y: {randomY()}; opacity: 1 }
  to   { opacity: 0 }
  over: 500ms
  loop: infinite
  repeatRefresh: true          // 每次循环重新计算 randomX() / randomY()
}
