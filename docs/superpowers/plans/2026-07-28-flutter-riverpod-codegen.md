# Flutter Codegen: ChangeNotifier → Riverpod 3.0

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace ChangeNotifier-based Flutter codegen with Riverpod 3.0 provider-based codegen for stores and components.

**Architecture:** Each `.omni` store becomes a `*_store.dart` file with Riverpod `StateProvider`/`Notifier` declarations. Components that reference stores become `ConsumerStatefulWidget` (replacing `StatefulWidget`) and use `ref.watch(provider)` for reads and `ref.read(notifierProvider.notifier).method()` for mutations. Component-local signals remain as direct field assignments (no Riverpod needed for local state). IR layer is untouched — all changes are in the Flutter codegen backend.

**Tech Stack:** Rust (codegen), Dart/Riverpod 3.0 (generated output), `flutter_riverpod` package (runtime dependency of generated code)

## Global Constraints

- Only `omnilang/src/codegen/flutter/` directory is modified; IR layer untouched
- ADR-003: Flutter is P0 target, full test coverage required
- ADR-020: Riverpod 3.0 is the chosen state management library
- Generated code must pass `dart analyze` static checks (non-blocking for Rust CI)
- All existing Flutter tests must pass after update (updated assertions acceptable)
- Parameterized regression tests (`tests_regression/parameterized.rs`) run per-backend; snapshot files must be reviewed/updated

fn gen_store_dart(&self, name: &str, store: &StoreIR) -> String {
    let mut w = CodeWriter::new();
    w.emit_fmt(format_args!(
        "// Generated by OmniLang \u{2014} Store: {}",
        name
    ));
    w.emit("");

    // Always import flutter_riverpod for provider types
    w.emit("import 'package:flutter_riverpod/flutter_riverpod.dart';");
    w.emit("");

    // ── State class (holds all signals as mutable fields) ──
    let state_class = format!("{}State", name);
    w.emit_fmt(format_args!("class {} {{", state_class));
    w.indent();
    for (sig_name, sig_type) in &store.signals {
        let ann = super::dart::dart_type(sig_type.kind.as_str(), &sig_type.type_args);
        w.emit_fmt(format_args!("final {} {};", ann, sig_name));
    }
    // Constructor
    w.emit("");
    w.emit_fmt(format_args!("const {}({{", state_class));
    w.indent();
    for (sig_name, sig_type) in &store.signals {
        let ann = super::dart::dart_type(sig_type.kind.as_str(), &sig_type.type_args);
        let default_val = store
            .signal_defaults
            .get(sig_name)
            .map(|v| format_json_value(v, &JSON_FLUTTER))
            .unwrap_or_else(|| "null".into());
        w.emit_fmt(format_args!("this.{} = {},", sig_name, default_val));
    }
    w.dedent();
    w.emit("});");
    w.dedent();
    w.emit("}");

    // ── Notifier class (holds actions) ──
    let notifier_class = format!("{}Notifier", name);
    let notifier_provider_name = format!("{}NotifierProvider", name);
    let state_provider_base = format!("{}StateProvider", name);

    // Emit a typedef for readability
    w.emit("");
    w.emit_fmt(format_args!(
        "typedef {} = AutoDisposeNotifierProvider<{}, {}>;",
        notifier_provider_name, notifier_class, state_class
    ));

    // Notifier class
    w.emit("");
    w.emit_fmt(format_args!(
        "class {} extends AutoDisposeNotifier<{}> {{",
        notifier_class, state_class
    ));
    w.indent();

    // build() — initial state from defaults
    w.emit("@override");
    w.emit_fmt(format_args!("{} build() {{", state_class));
    w.indent();
    w.emit_fmt(format_args!("return {}({{", state_class));
    w.indent();
    for sig_name in store.signals.keys() {
        let default_val = store
            .signal_defaults
            .get(sig_name)
            .map(|v| format_json_value(v, &JSON_FLUTTER))
            .unwrap_or_else(|| "null".into());
        w.emit_fmt(format_args!("{}: {},", sig_name, default_val));
    }
    w.dedent();
    w.emit("});");
    w.dedent();
    w.emit("}");

    // Actions
    for action in &store.actions {
        w.emit("");
        let params: Vec<String> = action
            .params
            .iter()
            .map(|p| {
                let ty = p
                    .r#type
                    .as_ref()
                    .map(|t| super::dart::dart_type(t.kind.as_str(), &t.type_args))
                    .unwrap_or_else(|| "dynamic".into());
                format!("{} {}", ty, p.name)
            })
            .collect();
        let prefix = if action.is_async {
            "Future<void>"
        } else {
            "void"
        };
        w.emit_fmt(format_args!(
            "{} {}({}) {{",
            prefix,
            action.name,
            params.join(", ")
        ));
        w.indent();
        for stmt in &action.body.statements {
            generate_statement_flutter(&mut w, stmt);
        }
        w.dedent();
        w.emit("}");
    }

    w.dedent();
    w.emit("}");

    // ── Top-level provider instance ──
    w.emit("");
    w.emit_fmt(format_args!(
        "final {} = NotifierProvider<{}, {}>({}_new);",
        notifier_provider_name.to_lowercase(), notifier_class, state_class, name
    ));

    // ── Getters as derived providers ──
    for (getter_name, getter_expr) in &store.getters {
        let getter_provider_name = format!("{}{}", name, capitalize_first(getter_name));
        let body = format_expr_ir_flutter(getter_expr);
        w.emit("");
        w.emit_fmt(format_args!(
            "final {} = Provider<dynamic>((ref) {{",
            getter_provider_name
        ));
        w.indent();
        // Read the state to establish dependency
        w.emit_fmt(format_args!(
            "final _state = ref.watch({});",
            notifier_provider_name.to_lowercase()
        ));
        w.emit_fmt(format_args!("return _state.{};", body));
        w.dedent();
        w.emit("});");
    }

    // ── Convenience providers for individual signal reads ──
    for (sig_name, sig_type) in &store.signals {
        let sig_provider_name = format!("{}{}Provider", name, capitalize_first(sig_name));
        let ann = super::dart::dart_type(sig_type.kind.as_str(), &sig_type.type_args);
        w.emit("");
        w.emit_fmt(format_args!(
            "final {} = Provider<{}>((ref) {{",
            sig_provider_name, ann
        ));
        w.indent();
        w.emit_fmt(format_args!(
            "return ref.watch({}).{};",
            notifier_provider_name.to_lowercase(), sig_name
        ));
        w.dedent();
        w.emit("});");
    }

    w.into_string()
}

fn gen_store_dart(&self, name: &str, store: &StoreIR) -> String {
    let mut w = CodeWriter::new();
    w.emit_fmt(format_args!(
        "// Generated by OmniLang \u{2014} Store: {}", name
    ));
    w.emit("");

    if store.signals.is_empty() && store.getters.is_empty() {
        w.emit("import 'package:flutter_riverpod/flutter_riverpod.dart';");
        w.emit("");
        w.emit("// Empty store — no signals, getters, or actions to emit");
        return w.into_string();
    }

    w.emit("import 'package:flutter_riverpod/flutter_riverpod.dart';");
    w.emit("");

    let lower = |s: &str| {
        let mut c = s.chars();
        match c.next() {
            None => String::new(),
            Some(first) => first.to_lowercase().to_string() + c.as_str(),
        }
    };
    let store_prefix = lower(name);

    // ── Signal providers ──
    if !store.signals.is_empty() {
        w.emit("// ── Signal providers ──");
        for (sig_name, sig_type) in &store.signals {
            let provider_name = format!("{}{}Provider", store_prefix, capitalize_first(sig_name));
            let ann = super::dart::dart_type(sig_type.kind.as_str(), &sig_type.type_args);
            let default_val = store
                .signal_defaults
                .get(sig_name)
                .map(|v| format_json_value(v, &JSON_FLUTTER))
                .unwrap_or_else(|| "null".into());
            w.emit_fmt(format_args!(
                "final {} = StateProvider<{}>((ref) => {});",
                provider_name, ann, default_val
            ));
        }
        w.emit("");
    }

    // ── Getter providers ──
    for (getter_name, getter_expr) in &store.getters {
        let provider_name = format!("{}{}Provider", store_prefix, capitalize_first(getter_name));
        let body_dart = format_expr_ir_flutter(getter_expr);
        w.emit_fmt(format_args!(
            "final {} = Provider<dynamic>((ref) {{", provider_name
        ));
        w.indent();
        // Emit ref.watch() for each signal so the getter auto-tracks dependencies
        for sig_name in store.signals.keys() {
            let sig_provider = format!("{}{}Provider", store_prefix, capitalize_first(sig_name));
            w.emit_fmt(format_args!("ref.watch({});", sig_provider));
        }
        w.emit_fmt(format_args!("return {};", body_dart));
        w.dedent();
        w.emit("});");
        w.emit("");
    }

    // ── Action functions ──
    if !store.actions.is_empty() {
        w.emit("// ── Action functions ──");
        for action in &store.actions {
            let action_name = format!("{}{}", store_prefix, capitalize_first(&action.name));
            let mut params: Vec<String> = vec!["WidgetRef ref".to_string()];
            for p in &action.params {
                let ty = p.r#type
                    .as_ref()
                    .map(|t| super::dart::dart_type(t.kind.as_str(), &t.type_args))
                    .unwrap_or_else(|| "dynamic".into());
                params.push(format!("{} {}", ty, p.name));
            }
            let prefix = if action.is_async { "Future<void>" } else { "void" };
            w.emit_fmt(format_args!(
                "{} {}({}) {{", prefix, action_name, params.join(", ")
            ));
            w.indent();
            for stmt in &action.body.statements {
                generate_statement_flutter(&mut w, stmt);
            }
            w.dedent();
            w.emit("}");
            w.emit("");
        }
    }

    w.into_string()
}

pub(crate) struct FlutterStmtFormatter {
    pub(crate) expr: DartFormatExpr,
    /// When true, mutations use Riverpod state mutation pattern
    /// instead of setState wrapping.
    pub(crate) in_store_action: bool,
    /// Store prefix for constructing provider names (e.g. "appStore")
    pub(crate) store_prefix: Option<String>,
}

impl FormatStmt for FlutterStmtFormatter {
    // ... existing methods unchanged ...

    fn mutation_stmt_prefix(&self) -> Option<&str> {
        if self.in_store_action {
            None  // No setState wrapping in store actions
        } else {
            Some("setState(() {")
        }
    }

    fn mutation_stmt_suffix(&self) -> Option<&str> {
        if self.in_store_action {
            None  // No setState wrapping in store actions
        } else {
            Some("});")
        }
    }

    fn mutation_target_expr(&self, target: &str) -> String {
        if let Some(ref prefix) = self.store_prefix {
            // In store action: signal = ... → ref.read(prefixSignalProvider.notifier).state = ...
            format!("ref.read({}{}Provider.notifier).state", prefix, capitalize_first(target))
        } else {
            target.to_string()
        }
    }
}
